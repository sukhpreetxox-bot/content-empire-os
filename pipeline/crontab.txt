# Content Empire OS — crontab for the Oracle Cloud Always-Free VM
# Install with:  crontab pipeline/crontab.txt
# All times UTC. Adjust the venv path if you use a different location.
#
# Layout assumed:
#   /home/ubuntu/ContentEmpire           (git clone)
#   /home/ubuntu/ContentEmpire/.venv     (python -m venv .venv)

PROJECT=/home/ubuntu/ContentEmpire
PY=/home/ubuntu/ContentEmpire/.venv/bin/python

# 1) GENERATION — stagger YouTube vs Instagram so renders don't overlap.
#    07:00 UTC: YouTube channels   |   07:30 UTC: Instagram channels
0  7 * * *  cd $PROJECT && $PY pipeline/gen_cron.py --platform youtube   >> $PROJECT/logs/gen_yt.log 2>&1
30 7 * * *  cd $PROJECT && $PY pipeline/gen_cron.py --platform instagram >> $PROJECT/logs/gen_ig.log 2>&1

# 2) PUBLICATION — runs hourly during the day; only picks APPROVED & due rows,
#    respects daily caps + throttling internally. (You approve via the dashboard.)
0  8-20 * * *  cd $PROJECT && $PY pipeline/publish_cron.py >> $PROJECT/logs/publish.log 2>&1

# 3) ANALYTICS — once daily; also keeps Supabase awake.
0  23 * * *  cd $PROJECT && $PY pipeline/analytics_cron.py >> $PROJECT/logs/analytics.log 2>&1

# Optional extra keep-alive ping (lightweight) every 3 days at noon.
0 12 */3 * *  cd $PROJECT && $PY -c "import sys; sys.path.insert(0,'pipeline'); from helpers import db; db.touch_keepalive()" >> $PROJECT/logs/keepalive.log 2>&1
